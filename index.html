async function startScan() { document.getElementById("reader").style.display = "block"; try { if (scanner) { await scanner.clear(); scanner = null; } scanner = new Html5Qrcode("reader"); document.getElementById("status").innerText = "AVVIO CAMERA..."; await scanner.start( { facingMode: { ideal: "environment" } }, { fps: 10, qrbox: 250 }, (text) => { console.log("QR LETTO:", text); } ); document.getElementById("status").innerText = "CAMERA ATTIVA"; } catch (err) { document.getElementById("status").innerHTML = "ERRORE CAMERA: " + err + ""; console.log(err); } }
