async function startScan() { document.getElementById("status").innerText = "AVVIO CAMERA..."; try { scanner = new Html5Qrcode("reader"); const devices = await Html5Qrcode.getCameras(); if (!devices || devices.length === 0) { throw "NESSUNA CAMERA TROVATA"; } // 🔥 CERCA CAMERA POSTERIORE let backCamera = devices.find(d => d.label.toLowerCase().includes("back") || d.label.toLowerCase().includes("rear") || d.label.toLowerCase().includes("environment") ); // fallback if (!backCamera) { backCamera = devices[0]; } await scanner.start( { deviceId: backCamera.id }, { fps: 10, qrbox: 250 }, (text) => { document.getElementById("status").innerText = text; } ); document.getElementById("status").innerText = "SCANSIONE ATTIVA"; } catch (err) { document.getElementById("status").innerText = "CAMERA NON ATTIVA: " + err; } }
